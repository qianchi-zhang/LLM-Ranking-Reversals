\begin{frame}{A Familiar Story, and the Question Behind It}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{beamercolorbox}[rounded=true,shadow=true,sep=0.9em,wd=\linewidth]{block body}
        {\large \textbf{What a leaderboard suggests}}\\[0.5em]
        \begin{itemize}
          \item Model A: 89.2\%
          \item Model B: 87.3\%
          \item Therefore, A is better than B
        \end{itemize}
      \end{beamercolorbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{beamercolorbox}[rounded=true,shadow=true,sep=0.9em,wd=\linewidth]{block body}
        {\large \textbf{What we ask instead}}\\[0.5em]
        \begin{itemize}
          \item What if we ask the same question differently?
          \item What if we evaluate on a slightly different subset?
          \item Does the rank order remain the same?
        \end{itemize}
      \end{beamercolorbox}
    \end{column}
  \end{columns}

  \vspace{0.9em}
  \insightbox{Our target is not only score accuracy, but the reliability of the ranking itself.}
\end{frame}

\begin{frame}{Motivation}
  \begin{itemize}
    \item Static leaderboards are attractive because they compress a benchmark into a single ordering.
    \item But that ordering implicitly assumes \textbf{evaluation stability}: small, reasonable protocol changes should not alter the winner.
    \item In practice, benchmark outcomes depend on at least two nontrivial factors:
    \begin{itemize}
      \item prompt formulation
      \item evaluation subset composition
    \end{itemize}
    \item If these sources of variation are large relative to model gaps, rank claims may be overconfident.
  \end{itemize}

  \vspace{0.8em}
  \insightbox{We study when a leaderboard is robust, and when it is only an artifact of one evaluation choice.}
\end{frame}

\begin{frame}{Positioning Relative to Prior Work}
  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \metriccard{Benchmark papers}{MMLU, ARC, and HellaSwag establish the task settings we evaluate.}
    \end{column}
    \begin{column}{0.31\textwidth}
      \metriccard{Prompt robustness}{Prior work shows that task-preserving prompt changes can affect LLM behavior.}
    \end{column}
    \begin{column}{0.31\textwidth}
      \metriccard{Evaluation reliability}{Recent discussions emphasize that protocol matters, not just model identity.}
    \end{column}
  \end{columns}

  \vspace{0.9em}
  \begin{block}{Our contribution}
    We focus on \textbf{ranking stability}: not just whether scores move, but whether the order between models reverses.
  \end{block}
\end{frame}

\begin{frame}{Research Questions}
  \begin{enumerate}
    \item Do non-semantic prompt perturbations change model performance?
    \item Does evaluation-subset variation change relative model rankings?
    \item Are those changes large enough to create \textbf{ranking reversals}?
  \end{enumerate}

  \vspace{0.9em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{beamercolorbox}[rounded=true,shadow=true,sep=0.8em,wd=\linewidth]{block body}
        {\large \textbf{If the answer is no}}\\[0.4em]
        A static leaderboard is a reasonable summary.
      \end{beamercolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{beamercolorbox}[rounded=true,shadow=true,sep=0.8em,wd=\linewidth]{block body}
        {\large \textbf{If the answer is yes}}\\[0.4em]
        Rank claims should be reported with uncertainty and robustness diagnostics.
      \end{beamercolorbox}
    \end{column}
  \end{columns}
\end{frame}

\sectionframe{Method}{Treat ranking as a random outcome induced by evaluation choices}

\begin{frame}{Core Idea and Theoretical Framing}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item Keep task semantics fixed.
        \item Perturb the prompt in a controlled, non-semantic way.
        \item Resample the evaluation subset via bootstrap.
        \item Recompute the ranking under each induced evaluation configuration.
      \end{itemize}

      \vspace{0.5em}
      \[
      r_{d,p,b}(m) = \text{rank of model } m
      \]
      \[
      \text{RRR}_{d,p} = \Pr \left( r_{d,p,b} \neq r_{d,\text{minimal},b} \right)
      \]
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{beamercolorbox}[rounded=true,shadow=true,sep=0.9em,wd=\linewidth]{block body}
        {\large \textbf{Interpretation}}\\[0.5em]
        Rank is not treated as a fixed truth, but as a distribution over plausible evaluation choices.
      \end{beamercolorbox}
    \end{column}
  \end{columns}

  \vspace{0.8em}
  \insightbox{High RRR means a single leaderboard is a fragile snapshot rather than a stable conclusion.}
\end{frame}

\sectionframe{Takeaways}{What this project says about how LLM evaluation should be reported}

\begin{frame}{What We Learned}
  \begin{enumerate}
    \item Prompt perturbations and subset variation do not affect all benchmarks equally.
    \item Ranking reversals occur most often when performance gaps are narrow.
    \item A single leaderboard may be robust, partially stable, or fragile depending on the task.
    \item Average accuracy alone is not enough; rank uncertainty deserves explicit reporting.
  \end{enumerate}

  \vspace{0.9em}
  \insightbox{A better benchmark report should state not only who is best on average, but how robust that ordering is.}
\end{frame}

\begin{frame}{Implications and Limitations}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Implications}
      \begin{itemize}
        \item report confidence intervals and pairwise significance
        \item report ranking stability metrics such as RRR
        \item avoid overinterpreting narrow score gaps as stable superiority
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Limitations}
      \begin{itemize}
        \item only 3 multiple-choice benchmarks
        \item only 4 models
        \item prompt perturbations are controlled and shallow, not exhaustive
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.9em}
  \begin{beamercolorbox}[rounded=true,shadow=true,sep=0.8em,wd=\linewidth]{block body}
    This is not a claim that all leaderboards are unreliable. It is a claim that reliability must be demonstrated, not assumed.
  \end{beamercolorbox}
\end{frame}
